📝 EXEMPLO DE USO NO AUTHCONTROLLER:

import { ErrorResponse } from '../commom/response/errorResponse';
import { ErrorEnum } from '../commom/enum/error.enum';
import { ThrottleStrict } from '../commom/decorators/throttle.decorator';
import { BruteForceGuard } from '../commom/guards/brute-force.guard';

@Controller('auth')
export class AuthController {
  constructor(
    private authService: AuthService,
    private bruteForceGuard: BruteForceGuard,
  ) {}

  @Post('login')
  @Public()
  @ThrottleStrict() // 5 req/min
  @UseGuards(BruteForceGuard)
  async login(@Body() loginDto: LoginAuthDto, @Req() req) {
    const ip = this.getClientIp(req);

    try {
      const result = await this.authService.login(loginDto);
      this.bruteForceGuard.clearAttempts(ip, loginDto.email);
      return result;
    } catch (error) {
      this.bruteForceGuard.registerSuspiciousActivity(ip);
      throw new ErrorResponse(error);
    }
  }

  private getClientIp(req: any): string {
    return req.headers['x-forwarded-for']?.split(',')[0] || 
           req.headers['x-real-ip'] || 
           req.ip || 
           'unknown';
  }
}

📱 TRATAMENTO NO FRONTEND:

// React Native / Next.js
try {
  await api.post('/auth/login', data);
} catch (error) {
  const errCode = error.response?.data?.errorsCode;
  const message = error.response?.data?.message;

  switch (errCode) {
    case 'E-0007': // Rate limit
      Alert.alert('Aguarde', message);
      break;
    case 'E-0010': // Conta bloqueada
      Alert.alert('Bloqueado', message);
      break;
    case 'E-0008': // IP bloqueado
      Alert.alert('Acesso Negado', message);
      break;
    default:
      Alert.alert('Erro', message);
  }
}
